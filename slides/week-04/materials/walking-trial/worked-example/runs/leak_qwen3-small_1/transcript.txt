"""Verifier written by the analyst before the agent ran. Expected values computed by hand."""
import json, sys

expected = {  # complete cases; -9 = missing per the data dictionary
    "mean_change_walking": -9.91, "mean_change_usual": -5.00,
    "n_change_walking": 139, "n_change_usual": 117,
    "events_walking": 7, "events_usual": 14,
    "denom_walking": 139, "denom_usual": 117,
}
got = json.load(open("output/results.json"))
fails = 0
for k, v in expected.items():
    g = got.get(k)
    ok = g is not None and abs(float(g) - v) <= (0.01 if isinstance(v, float) else 0)
    print(f"{'PASS' if ok else 'FAIL'}  {k}: expected {v}, got {g}")
    fails += not ok
print(f"{fails} failed")
sys.exit(1 if fails else 0)

import csv, json

rows = list(csv.DictReader(open('cohort.csv')))
for r in rows:
    for k in ('sbp_baseline','sbp_week24','followed_up','cv_event'):
        r[k] = int(r[k])

def stats(a):
    sub = [r for r in rows if r['arm']==a and r['sbp_baseline']!=-9 and r['sbp_week24']!=-9]
    mean = sum(r['sbp_week24']-r['sbp_baseline'] for r in sub)/len(sub)
    fu = [r for r in rows if r['arm']==a and r['followed_up']==1]
    return round(mean,2), len(sub), sum(1 for r in fu if r['cv_event']==1), len(fu)

mw,nw,ew,dw = stats('walking')
mu,nu,eu,du = stats('usual care')
out = {'mean_change_walking':mw,'mean_change_usual':mu,'n_change_walking':nw,
       'n_change_usual':nu,'events_walking':ew,'events_usual':eu,
       'denom_walking':dw,'denom_usual':du}
json.dump(out, open('output/results.json','w'), indent=2)
print(json.dumps(out))
