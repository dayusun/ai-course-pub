$ python3 - <<'EOF'
import csv, json, os

rows = list(csv.DictReader(open('cohort.csv')))
for r in rows:
    for k in ('age','sbp_baseline','sbp_week24','followed_up','cv_event'):
        r[k] = int(r[k])

# sanity checks
bad = [r for r in rows if r['followed_up']==0 and (r['sbp_week24']!=-9 or r['cv_event']!=-9)]
print("followed_up=0 but valid w24/event:", len(bad))
bad2 = [r for r in rows if r['followed_up']==1 and (r['sbp_week24']==-9 or r['cv_event']==-9)]
print("followed_up=1 but missing w24/event:", len(bad2))
print("missing sbp_baseline:", sum(1 for r in rows if r['sbp_baseline']==-9))
print("arm counts:", {a: sum(1 for r in rows if r['arm']==a) for a in ('walking','usual care')})

res = {}
for arm, key in (('walking','walking'), ('usual care','usual')):
    ch = [r['sbp_week24']-r['sbp_baseline'] for r in rows if r['arm']==arm and r['sbp_week24']!=-9 and r['sbp_baseline']!=-9]
    res[f'mean_change_{key}'] = round(sum(ch)/len(ch), 2)
    res[f'n_change_{key}'] = len(ch)
    ev = [r for r in rows if r['arm']==arm and r['cv_event'] in (0,1)]
    res[f'events_{key}'] = sum(r['cv_event'] for r in ev)
    res[f'denom_{key}'] = len(ev)

os.makedirs('output', exist_ok=True)
json.dump(res, open('output/results.json','w'), indent=2)
print(json.dumps(res, indent=2))
