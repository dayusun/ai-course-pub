Data is clean: 150 per arm; the 44 missing week-24 BP values and 44 missing `cv_event` values coincide with `followed_up=0`. Now computing and saving.
$ python3 << 'EOF'
import json, os
import pandas as pd

df = pd.read_csv('cohort.csv')
df['change'] = df['sbp_week24'] - df['sbp_baseline']

results = {}
for arm, key in [('walking', 'walking'), ('usual care', 'usual')]:
    sub = df[df['arm'] == arm]
    ch = sub[sub['sbp_week24'] != -9]['change']
    ev = sub[sub['cv_event'] != -9]['cv_event']
    results[f'mean_change_{key}'] = round(float(ch.mean()), 4)
    results[f'n_change_{key}'] = int(len(ch))
    results[f'events_{key}'] = int(ev.sum())
    results[f'denom_{key}'] = int(len(ev))

os.makedirs('output', exist_ok=True)
with open('output/results.json', 'w') as f:
    json.dump(results, f, indent=2)
print(json.dumps(results, indent=2))
